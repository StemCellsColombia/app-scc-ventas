function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function doPost(e) {
  try {
    var p = (e && e.parameter) || {}, body = {};
    try { body = JSON.parse((e.postData && e.postData.contents) || '{}'); } catch (x) { body = {}; }
    if (p.hook) return json_(webhook_(p, body));
    return json_(api_(body));
  } catch (err) {
    return json_({ ok: false, error: String(err.message || err) });
  }
}

/* ------------------------------------------------------------------ identidad */
function quien_(idToken) {
  if (!idToken) throw new Error('Sin sesión');
  var c = CacheService.getScriptCache(), k = 'id_' + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, idToken)).slice(0, 40);
  var correo = c.get(k);
  if (!correo) {
    var r = UrlFetchApp.fetch('https://identitytoolkit.googleapis.com/v1/accounts:lookup?key=' + APP_.API_KEY, {
      method: 'post', contentType: 'application/json', payload: JSON.stringify({ idToken: idToken }), muteHttpExceptions: true });
    var u = r.getResponseCode() === 200 && (JSON.parse(r.getContentText()).users || [])[0];
    if (!u || !u.email) throw new Error('Sesión vencida: vuelve a entrar');
    correo = String(u.email).toLowerCase();
    c.put(k, correo, 3000);
  }
  var m = EQUIPO_[correo];
  if (!m) throw new Error('Este correo no tiene acceso a la app');
  return { correo: correo, nombre: m.nombre, admin: m.admin, ghl: m.ghl };
}

/* Nombre con el que aparece en GHL (el Radar agrupa por el primer nombre del usuario de GHL). */
function nombreGhl_(yo, usuarios) {
  var ids = Object.keys(usuarios || {});
  for (var i = 0; i < ids.length; i++) {
    var n = norm(usuarios[ids[i]]);
    for (var j = 0; j < yo.ghl.length; j++) if (n.indexOf(yo.ghl[j]) === 0) return usuarios[ids[i]];
  }
  return '';
}

/* ------------------------------------------------------------------ API de la app */
function api_(b) {
  var yo = quien_(b.idToken);
  var R;
  switch (b.accion) {
    case 'radar':
      R = radarPara_(yo, !!b.forzar);
      return { ok: true, yo: R.yo, radar: R };
    case 'marcar':
      if (!ACCIONES_OK_[String(b.que || '').replace('deshacer:', '')]) throw new Error('Acción no válida');
      var lock = LockService.getScriptLock(); lock.waitLock(10000);
      var ghl = '';
      try { ghl = escribirGhl_(b.opp, b.contactId, b.que, yo.nombre); } catch (e) { ghl = 'error: ' + e.message; }
      try { hoja_().appendRow([new Date(), yo.nombre, b.opp, b.contactId, b.que, b.nombre || '', ghl]); } finally { lock.releaseLock(); }
      R = radarPara_(yo, false);
      return { ok: true, yo: R.yo, radar: R };
    case 'token':
      guardarToken_(yo.correo, String(b.token || ''), String(b.equipo || ''));
      return { ok: true };
    case 'probar':
      var n = push_([yo.correo], 'Avisos activos', 'Así te va a sonar cuando un paciente necesite que lo llames.', APP_.URL, 'prueba');
      return { ok: true, enviados: n };
    case 'nota':
      return { ok: true, ghl: guardarNota_(String(b.opp || ''), String(b.c || ''), b.nota, b.prox, String(b.fecha || ''), yo) };
    case 'tablero':
      return { ok: true, tablero: tablero_(!!b.forzar) };
    case 'atendido':
      marcarEvento_(String(b.id || ''), yo.nombre, !!b.deshacer);
      R = radarPara_(yo, false);
      return { ok: true, yo: R.yo, radar: R };
    // Pendientes (14-oct): cada escritura devuelve las listas completas para que todos los aparatos queden iguales
    case 'tareas_listar':
    case 'notas_listar':
      return pendientes_(yo, {});
    case 'tarea_guardar':
      return pendientes_(yo, { tarea: tareaGuardar_(b.tarea || {}, yo) });
    case 'tarea_borrar':
      pendBorrar_('tareas', String(b.id || ''), yo);
      return pendientes_(yo, {});
    case 'nota_guardar':
      return pendientes_(yo, { nota: notaRapidaGuardar_(b.nota || {}, yo) });
    case 'nota_borrar':
      pendBorrar_('notas', String(b.id || ''), yo);
      return pendientes_(yo, {});
    default:
      throw new Error('Acción desconocida');
  }
}

function radarPara_(yo, forzar) {

  raw.ahora = Date.now();
  var R = armarRadar(raw);
  R.yo = { nombre: yo.nombre, admin: yo.admin, ghl: nombreGhl_(yo, raw.usuarios) };
  // idioma y condición de cada tarjeta (de las etiquetas del contacto)
  var porOpp = {};
  (raw.opps || []).forEach(function (o) { porOpp[o.id] = o; });
  R.secciones.forEach(function (s) {
    s.items.forEach(function (x) {
      var tags = ((porOpp[x.opp] || {}).tags || []).map(function (t) { return String(t).toLowerCase(); });

  return R;
}

/* ------------------------------------------------------------------ TABLERO (14-oct, Mia): pacientes por semana desde el Calendar de info@
   La Scheduling crea en el calendario principal de info@ eventos "NOMBRE - <qué>" (ver scheduling/index.html, buildEvents).
   - Tratamiento: Incodol, QUIROFANO, Sueroterapia, Exosomas, Facial, Capilar, CO2, Hyperbaric, VITAL, Vitamin C, Essenza, Meet with Patient Care…
   - «Listo» = el paciente ya tiene "Transfer - Aeropuerto Llegada" (mandó el vuelo). Si no, «Pendiente».
   - Se ignoran follow-ups, consultas online, recordatorios internos (⚠️ / 📞) y marcadores tumorales.
   Devuelve solo nombre, días y tipo de tratamiento; nada de correos, teléfonos ni descripciones. Caché de 10 min. */
var TB_SEMANAS_ = 8;
function semanaIso_(ymd) {                                                             // número de semana ISO (lunes a domingo)
  var d = new Date(ymd + 'T12:00:00Z'); d.setUTCDate(d.getUTCDate() - ((d.getUTCDay() + 6) % 7) + 3);
  var y = new Date(Date.UTC(d.getUTCFullYear(), 0, 4));
  return 1 + Math.round(((d - y) / 86400000 - 3 + ((y.getUTCDay() + 6) % 7)) / 7);
}
function tablero_(forzar) {
  var cache = CacheService.getScriptCache(), k = 'tablero_v1';
  if (!forzar) { var c = cache.get(k); if (c) return JSON.parse(c); }
  var TZ = 'America/Bogota', DIA_ = 86400000;
  var hoy = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');
  var dow = (new Date(hoy + 'T12:00:00Z').getUTCDay() + 6) % 7;                      // 0 = lunes
  var lunes = new Date(new Date(hoy + 'T00:00:00-05:00').getTime() - dow * DIA_);
  var fin = new Date(lunes.getTime() + TB_SEMANAS_ * 7 * DIA_);
  var desde = new Date(lunes.getTime() - 10 * DIA_);                                  // la llegada puede ser antes del lunes
  var IGNORAR = /follow up|online consultation|cancer markers|revisar|lectura|agendar|aprobaci/i;
  var CLINICO = /incodol|quirofano|quirófano|sueroterapia|premedicaci|blood detox|exosom|facial|capilar|co2|laser|láser|hyperbaric|vital|vitamin c|essenza|patient care|c[eé]lulas|stem cell|limpieza/i;
  var P = {};
  CalendarApp.getDefaultCalendar().getEvents(desde, fin).forEach(function (ev) {
    var t = String(ev.getTitle() || '').trim();
    if (!t || /^(⚠|📞)/.test(t) || IGNORAR.test(t)) return;
    var i = t.indexOf(' - '); if (i < 2) return;
    var nombre = t.slice(0, i).trim(), resto = t.slice(i + 3), key = norm(nombre);
    if (!key || nombre.length > 60) return;
    var p = P[key] = P[key] || { n: nombre, llegada: false, dias: {}, tx: {} };
    if (/aeropuerto llegada/i.test(resto)) { p.llegada = true; return; }
    if (!CLINICO.test(resto)) return;
    var d = Utilities.formatDate(ev.getStartTime(), TZ, 'yyyy-MM-dd');
    if (ev.getStartTime() < lunes) return;
    p.dias[d] = 1;
    p.tx[/quirof/i.test(resto) ? 'Quirófano' : /hyperbaric/i.test(resto) ? 'Hiperbárica' : /facial|capilar|co2|laser|láser|essenza|limpieza/i.test(resto) ? 'Estética'
         : /exosom/i.test(resto) ? 'Exosomas' : /sueroterapia|premedicaci|detox|vitamin c|vital/i.test(resto) ? 'IV' : 'Otro'] = 1;
  });
  var semanas = [];
  for (var w = 0; w < TB_SEMANAS_; w++) {
    var ini = new Date(lunes.getTime() + w * 7 * DIA_);
    var iniTxt = Utilities.formatDate(ini, TZ, 'yyyy-MM-dd');
    semanas.push({ n: semanaIso_(iniTxt), ini: iniTxt, pac: [] });
  }
  Object.keys(P).forEach(function (key) {
    var p = P[key], dias = Object.keys(p.dias).sort();
    if (!dias.length) return;                                                          // sin tratamiento en la ventana: no es paciente de estas semanas
    var w = Math.floor((new Date(dias[0] + 'T12:00:00-05:00').getTime() - lunes.getTime()) / (7 * DIA_));
    if (w < 0 || w >= TB_SEMANAS_) return;
    var tx = Object.keys(p.tx); if (tx.length > 1) tx = tx.filter(function (x) { return x !== 'Otro'; });
    semanas[w].pac.push({ n: p.n, listo: p.llegada, dias: dias, tx: tx });
  });
  semanas.forEach(function (s) { s.pac.sort(function (a, b) { return a.dias[0] < b.dias[0] ? -1 : a.dias[0] > b.dias[0] ? 1 : 0; }); });
  var out = { generado: Date.now(), hoy: hoy, nSemanas: TB_SEMANAS_, semanas: semanas };
  try { cache.put(k, JSON.stringify(out), 600); } catch (e) {}
  return out;
}

function hojaNotas_() {
  var ss = hoja_().getParent(), h = ss.getSheetByName('notas');
  if (!h) { h = ss.insertSheet('notas'); h.appendRow(['Oportunidad', 'Nota', 'Próximo paso', 'Fecha próximo paso', 'Quién', 'Cuándo']); h.setFrozenRows(1); }

function leerNotas_() {
  var h = hojaNotas_(), n = h.getLastRow(), out = {};
  if (n < 2) return out;
  h.getRange(2, 1, n - 1, 6).getValues().forEach(function (r) {
    if (r[0]) out[String(r[0])] = { nota: String(r[1] || ''), prox: String(r[2] || ''), fecha: r[3] ? Utilities.formatDate(new Date(r[3]), 'America/Bogota', 'yyyy-MM-dd') : '',
                                     quien: String(r[4] || ''), at: r[5] ? new Date(r[5]).getTime() : 0 };
  });
  return out;
}
/* 13-oct (Mia y Alejo): las notas viven en GHL, donde las ve todo el equipo. La hoja solo guarda la última
   para pintar la Libreta rápido. En GHL: la nota va al contacto y el próximo paso con fecha queda como tarea del dueño. */
function guardarNota_(opp, contactId, nota, prox, fecha, yo) {
  if (!opp) throw new Error('Falta la oportunidad');
  nota = String(nota || '').slice(0, 2000); prox = String(prox || '').slice(0, 300);
  var antes = leerNotas_()[opp] || {};
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    var h = hojaNotas_(), n = h.getLastRow(), fila = 0;
    if (n >= 2) { var ids = h.getRange(2, 1, n - 1, 1).getValues(); for (var i = 0; i < ids.length; i++) if (String(ids[i][0]) === opp) { fila = i + 2; break; } }
    var v = [[opp, nota, prox, fecha ? new Date(fecha + 'T12:00:00-05:00') : '', yo.nombre, new Date()]];
    if (fila) h.getRange(fila, 1, 1, 6).setValues(v); else h.appendRow(v[0]);
  } finally { lock.releaseLock(); }
  return notaGhl_(contactId, nota, prox, fecha, yo, antes);
}

function notaGhl_(contactId, nota, prox, fecha, yo, antes) {
  if (!contactId) throw new Error('Se guardó en la app, pero falta el contacto para GHL');
  var h = { Authorization: 'Bearer ' + token_(), Version: CONFIG.VERSION, Accept: 'application/json' }, hecho = [];
  var cambioNota = nota && nota !== antes.nota, cambioPaso = prox && (prox !== antes.prox || fecha !== antes.fecha);
  if (cambioNota || cambioPaso) {
    var cuerpo = [cambioNota ? nota : '', prox ? 'Próximo paso: ' + prox + (fecha ? ' (' + fecha + ')' : '') : ''].filter(String).join('\n') +
                 '\n— ' + yo.nombre + ' · app SCC · ' + Utilities.formatDate(new Date(), 'America/Bogota', 'd MMM yyyy h:mm a');
    var r = UrlFetchApp.fetch(CONFIG.BASE + '/contacts/' + contactId + '/notes', { method: 'post', contentType: 'application/json', headers: h,
                              muteHttpExceptions: true, payload: JSON.stringify({ body: cuerpo }) });
    if (r.getResponseCode() >= 300) throw new Error('GHL no guardó la nota (' + r.getResponseCode() + ')');
    hecho.push('nota');
  }
  if (cambioPaso && fecha) {
    var raw = leerCache_() || {}, usuarios = raw.usuarios || {};
    var uid = Object.keys(usuarios).filter(function (id) { return yo.ghl.some(function (g) { return norm(usuarios[id]).indexOf(g) === 0; }); })[0];
    var tarea = { title: prox, body: nota.slice(0, 1000), dueDate: fecha + 'T14:00:00.000Z', completed: false };   // 9:00 am Colombia
    if (uid) tarea.assignedTo = uid;
    var t = UrlFetchApp.fetch(CONFIG.BASE + '/contacts/' + contactId + '/tasks', { method: 'post', contentType: 'application/json', headers: h,
                              muteHttpExceptions: true, payload: JSON.stringify(tarea) });
    if (t.getResponseCode() >= 300) throw new Error('La nota quedó en GHL, pero la tarea no (' + t.getResponseCode() + ')');
    hecho.push('tarea');
  }
  return hecho.join(' y ');
}
function libreta_(raw) {
  var notas = leerNotas_();
  return (raw.opps || []).filter(function (o) { return o.status === 'open'; }).map(function (o) {
    var c = (raw.conv || {})[o.contactId] || {}, tags = (o.tags || []).map(function (t) { return String(t).toLowerCase(); }), nt = notas[o.id] || {};
    return { opp: o.id, c: o.contactId, n: o.nombre || 'Sin nombre', et: raw.etapas[o.etapa] || '', tipo: tipoEtapa(raw.etapas[o.etapa]),
             dn: raw.usuarios[o.dueno] || 'Sin dueño', creado: o.creado, cambio: o.cambioEtapa || o.creado, tel: o.tel || '',
             es: tags.indexOf('es') >= 0, cond: condicionDe_(tags), ult: c.ultimoMsj || 0, dir: c.direccion || '', txt: c.ultimoTexto || '',
             nota: nt.nota || '', prox: nt.prox || '', fecha: nt.fecha || '', quien: nt.quien || '' };
  });
}

var CONDICIONES_ = ['knee', 'hip', 'shoulder', 'back', 'neck', 'ankle', 'hands', 'joint', 'face', 'facial', 'hair', 'sex', 'iv', 'kidney', 'diabetes', 'autoimmune', 'neuro', 'neuropathy'];
function condicionDe_(tags) {
  for (var i = 0; i < tags.length; i++) if (CONDICIONES_.indexOf(tags[i]) >= 0) return tags[i];
  return '';
}

function pendFilas_(h, ncol) {                                          // [{fila, d}] con d = el JSON de la columna «Datos»
  var n = h.getLastRow(), out = [];
  if (n < 2) return out;
  h.getRange(2, 1, n - 1, ncol).getValues().forEach(function (r, i) {
    var d = null; try { d = JSON.parse(String(r[ncol - 1] || '')); } catch (e) { d = null; }
    if (d && d.id) out.push({ fila: i + 2, d: d });
  });
  return out;
}
function limpio_(s, max) { return String(s == null ? '' : s).replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '').trim().slice(0, max); }
function celda_(s) { s = String(s == null ? '' : s); return /^[=+\-@]/.test(s) ? "'" + s : s; }   // texto que empieza con = no se vuelve fórmula

/* Crea o cambia una tarea. Solo se tocan los campos que llegan (así dos personas no se pisan). */
function tareaGuardar_(t, yo) {
  var paraOk = equipoNombres_().concat(['Todos']);
  var id = String(t.id || '');
  if (!/^t[a-z0-9]{6,24}$/.test(id)) id = 't' + Utilities.getUuid().replace(/-/g, '').slice(0, 12);
  var h = pendHoja_('tareas'), x = null, f = null, antesPara = null, nueva = false;
  var lock = LockService.getScriptLock(); lock.waitLock(15000);
  try {
    var filas = pendFilas_(h, PEND_.TAREAS.length);
    for (var i = 0; i < filas.length; i++) if (filas[i].d.id === id) { f = filas[i]; break; }
    if (f && f.d.borrada) return null;                                  // alguien la borró: no se revive
    if (f) { x = f.d; antesPara = x.para; }
    else {
      if (!('titulo' in t)) return null;                                // cambio a una tarea que no existe: se ignora
      x = { id: id, titulo: '', nota: '', para: yo.nombre, vence: '', prioridad: 'media', estado: 'pendiente',
            creadoPor: yo.nombre, creado: Date.now(), contactoId: '', contacto: '' };
      nueva = true;
    }
    if ('titulo' in t) { var s = limpio_(t.titulo, 200); if (!s) throw new Error('La tarea necesita un título'); x.titulo = s; }
    if ('nota' in t) x.nota = limpio_(t.nota, 2000);
    if ('para' in t) { if (paraOk.indexOf(t.para) < 0) throw new Error('«Para» no válido'); x.para = t.para; }
    if ('vence' in t) {
      var v = String(t.vence || '');
      if (v && !/^\d{4}-\d{2}-\d{2}$/.test(v)) throw new Error('Fecha no válida');
      x.vence = v;
    }
    if ('prioridad' in t) { if (PEND_.PRIORIDAD.indexOf(t.prioridad) < 0) throw new Error('Prioridad no válida'); x.prioridad = t.prioridad; }
    if ('estado' in t) {
      if (PEND_.ESTADO.indexOf(t.estado) < 0) throw new Error('Estado no válido');
      if (t.estado === 'hecho' && x.estado !== 'hecho') { x.hechoPor = yo.nombre; x.hecho = Date.now(); }
      if (t.estado !== 'hecho') { delete x.hechoPor; delete x.hecho; }
      x.estado = t.estado;
    }
    if ('contactoId' in t || 'contacto' in t) {
      var cid = String(t.contactoId || '');
      x.contactoId = /^[A-Za-z0-9]{6,40}$/.test(cid) ? cid : '';
      x.contacto = limpio_(t.contacto, 120);
    }
    x.cambio = Date.now(); x.cambioPor = yo.nombre;
    var fila = filaTarea_(x);
    if (f) h.getRange(f.fila, 1, 1, fila.length).setValues([fila]); else h.appendRow(fila);
  } finally { lock.releaseLock(); }
  if ((nueva || antesPara !== x.para) && x.estado !== 'hecho') { try { avisarTarea_(x, yo); } catch (e) { console.warn('Aviso de tarea: ' + e.message); } }
  return x;
}

/* Crea o cambia una nota rápida. */
function notaRapidaGuardar_(n, yo) {
  var id = String(n.id || '');
  if (!/^n[a-z0-9]{6,24}$/.test(id)) id = 'n' + Utilities.getUuid().replace(/-/g, '').slice(0, 12);
  var h = pendHoja_('notas'), x = null, f = null;
  var lock = LockService.getScriptLock(); lock.waitLock(15000);
  try {
    var filas = pendFilas_(h, PEND_.NOTAS.length);
    for (var i = 0; i < filas.length; i++) if (filas[i].d.id === id) { f = filas[i]; break; }
    if (f && f.d.borrada) return null;
    if (f) {
      x = f.d;
      if (!x.equipo && x.de !== yo.nombre) throw new Error('No puedes cambiar una nota personal de otra persona');
    } else {
      if (!('texto' in n)) return null;
      var porDefecto = { Alejo: 'aqua', Camilo: 'amarillo', Mia: 'rosa' }[yo.nombre] || 'amarillo';
      x = { id: id, texto: '', de: yo.nombre, equipo: true, fijada: false, color: porDefecto, creado: Date.now() };
    }
    if ('texto' in n) { var s = limpio_(n.texto, 2000); if (!s) throw new Error('La nota necesita texto'); x.texto = s; }
    if ('equipo' in n && !!n.equipo !== !!x.equipo) {
      if (x.de !== yo.nombre) throw new Error('Solo quien escribió la nota puede cambiar quién la ve');
      x.equipo = !!n.equipo;
    }
    if ('fijada' in n) x.fijada = !!n.fijada;
    if ('color' in n) { if (PEND_.COLORES.indexOf(n.color) < 0) throw new Error('Color no válido'); x.color = n.color; }
    x.cambio = Date.now(); x.cambioPor = yo.nombre;
    var fila = filaNota_(x);
    if (f) h.getRange(f.fila, 1, 1, fila.length).setValues([fila]); else h.appendRow(fila);
  } finally { lock.releaseLock(); }
  return x;
}

/* Borrar = marcar «borrada» (la fila se queda en la hoja). */
function pendBorrar_(hoja, id, yo) {
  var esTarea = hoja === 'tareas', cols = esTarea ? PEND_.TAREAS.length : PEND_.NOTAS.length;
  var h = pendHoja_(hoja);
  var lock = LockService.getScriptLock(); lock.waitLock(15000);
  try {
    var filas = pendFilas_(h, cols), f = null;
    for (var i = 0; i < filas.length; i++) if (filas[i].d.id === id) { f = filas[i]; break; }
    if (!f || f.d.borrada) return;
    var d = f.d, autor = esTarea ? d.creadoPor : d.de;
    if (!esTarea && !d.equipo && autor !== yo.nombre) throw new Error('No puedes borrar una nota personal de otra persona');
    if (autor !== yo.nombre && !yo.admin) {
      var pueden = [autor].concat(Object.keys(EQUIPO_).filter(function (c) { return EQUIPO_[c].admin && EQUIPO_[c].nombre !== autor; })
                                  .map(function (c) { return EQUIPO_[c].nombre; }));
      throw new Error('No puedes borrar lo que creó ' + autor + ' (solo lo borra ' + pueden.join(' o ') + ')');
    }
    d.borrada = Date.now(); d.borradaPor = yo.nombre; d.cambio = Date.now(); d.cambioPor = yo.nombre;
    var fila = esTarea ? filaTarea_(d) : filaNota_(d);
    h.getRange(f.fila, 1, 1, fila.length).setValues([fila]);
  } finally { lock.releaseLock(); }
}

/* Aviso al celular de quien recibe la tarea (no a quien la creó). De 9 pm a 7 am no suena. */
function avisarTarea_(t, yo) {
  var hora = Number(Utilities.formatDate(new Date(), 'America/Bogota', 'H'));
  if (hora < 7 || hora >= 21) return 0;
  var dest = Object.keys(EQUIPO_).filter(function (c) { return c !== yo.correo && (t.para === 'Todos' || EQUIPO_[c].nombre === t.para); });
  if (!dest.length) return 0;
  var cuando = t.vence ? ' · para el ' + Utilities.formatDate(new Date(t.vence + 'T12:00:00-05:00'), 'America/Bogota', 'd/M') : '';
  return push_(dest, (t.prioridad === 'alta' ? 'Tarea urgente de ' : 'Tarea nueva de ') + yo.nombre, t.titulo + cuando, APP_.URL + '#pendientes', 'tarea-' + t.id);
}

/* ------------------------------------------------------------------ señales de GHL (la fila "Ahora" de la app) */
function eventos_() {
  try { return JSON.parse(PropertiesService.getScriptProperties().getProperty('EVENTOS') || '[]'); } catch (e) { return []; }
}
function guardarEvento_(ev) {
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    var l = eventos_(), lim = Date.now() - 3 * 86400000;
    // una señal nueva del mismo paciente reemplaza la anterior sin atender
    l = l.filter(function (e) { return e.at > lim && !(e.c === ev.c && !e.d); });
    l.push(ev);
    while (JSON.stringify(l).length > 8500 && l.length) l.shift();   // límite de 9 KB por propiedad
    PropertiesService.getScriptProperties().setProperty('EVENTOS', JSON.stringify(l));
  } finally { lock.releaseLock(); }
}
function marcarEvento_(id, quien, deshacer) {
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    var l = eventos_();
    l.forEach(function (e) { if (e.id === id) { if (deshacer) { delete e.d; delete e.por; } else { e.d = Date.now(); e.por = quien; } } });
    PropertiesService.getScriptProperties().setProperty('EVENTOS', JSON.stringify(l));
  } finally { lock.releaseLock(); }
}

/* ------------------------------------------------------------------ tokens de avisos (uno por aparato) */
function tokens_() {
  try { return JSON.parse(PropertiesService.getScriptProperties().getProperty('FCM_TOKENS') || '{}'); } catch (e) { return {}; }
}
function guardarToken_(correo, token, equipo) {
  if (!token || token.length < 20) throw new Error('Token no válido');
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    var t = tokens_();
    Object.keys(t).forEach(function (c) { t[c] = (t[c] || []).filter(function (x) { return x.t !== token; }); });
    (t[correo] = t[correo] || []).push({ t: token, equipo: equipo.slice(0, 60), desde: Date.now() });
    t[correo] = t[correo].slice(-6);                                   // máximo 6 aparatos por persona
    PropertiesService.getScriptProperties().setProperty('FCM_TOKENS', JSON.stringify(t));
  } finally { lock.releaseLock(); }
}

/* Manda un aviso a esos correos (todos sus aparatos). Devuelve cuántos salieron. */
function push_(correos, titulo, cuerpo, link, tag) {
  var t = tokens_(), malos = [], n = 0;
  var oauth = ScriptApp.getOAuthToken();
  correos.forEach(function (c) {
    (t[c] || []).forEach(function (x) {
      var msg = { message: { token: x.t,
        data: { titulo: titulo, cuerpo: cuerpo, link: link || APP_.URL, tag: tag || 'scc' },
        webpush: { headers: { Urgency: 'high', TTL: '86400' },
                   notification: { title: titulo, body: cuerpo, icon: APP_.URL + 'icono-192.png', badge: APP_.URL + 'insignia-96.png',
                                   tag: tag || 'scc', renotify: true, requireInteraction: true },
                   fcm_options: { link: link || APP_.URL } } } };
      var r = UrlFetchApp.fetch('https://fcm.googleapis.com/v1/projects/' + APP_.PROYECTO + '/messages:send', {
        method: 'post', contentType: 'application/json', muteHttpExceptions: true, payload: JSON.stringify(msg),
        headers: { Authorization: 'Bearer ' + oauth, 'x-goog-user-project': APP_.PROYECTO } });
      var code = r.getResponseCode();
      if (code === 200) n++;
      else if (code === 404 || /UNREGISTERED|INVALID_ARGUMENT/.test(r.getContentText())) malos.push(x.t);
      else console.warn('FCM ' + code + ': ' + r.getContentText().slice(0, 300));
    });
  });
  if (malos.length) {
    Object.keys(t).forEach(function (c) { t[c] = (t[c] || []).filter(function (x) { return malos.indexOf(x.t) < 0; }); });
    PropertiesService.getScriptProperties().setProperty('FCM_TOKENS', JSON.stringify(t));
  }
  return n;
}

/* ------------------------------------------------------------------ avisos desde GHL (acción Webhook de los workflows) */
var AVISO_ = {
  nuevo:      ['Lead nuevo', 'Escríbele hoy desde el celular'],
  contesto:   ['Te contestó', 'Atiéndelo hoy'],
  cotizacion: ['Cotización enviada', 'Llámalo antes de 24 horas'],
  abrio:      ['Está mirando la cotización', 'Llámalo ya'],
  revision:   ['Pidió revisión de su caso', 'Llámalo ya y pídele la resonancia'],
  interes:    ['Le interesa', 'Tocó «Quiero saber más»: llámalo ya'],
  apartar:    ['Quiere confirmar esta semana', 'Llámalo ya y cierra fecha'],
  llamada:    ['Llamada perdida', 'Te llamó y nadie contestó: devuélvele la llamada']
};

function webhook_(p, b) {
  var clave = PropertiesService.getScriptProperties().getProperty('HOOK_CLAVE');
  if (!clave || p.hook !== clave) return { ok: false, error: 'clave' };
  var cd = b.customData || b.custom_data || {};
  var tipo = String(p.tipo || cd.tipo || b.tipo || '').toLowerCase();
  var a = AVISO_[tipo];
  if (!a) return { ok: false, error: 'tipo' };
  var contactId = b.contact_id || b.contactId || (b.contact && b.contact.id) || '';
  var nombre = b.full_name || [b.first_name, b.last_name].filter(String).join(' ') || (b.contact && b.contact.name) || 'Paciente';
  var duenoId = b.assigned_to || b.assignedTo || (b.contact && b.contact.assignedTo) || '';
  if (!duenoId && contactId) {
    try {
      var r = UrlFetchApp.fetch(CONFIG.BASE + '/contacts/' + contactId, { muteHttpExceptions: true,
        headers: { Authorization: 'Bearer ' + token_(), Version: CONFIG.VERSION, Accept: 'application/json' } });
      if (r.getResponseCode() === 200) duenoId = (JSON.parse(r.getContentText()).contact || {}).assignedTo || '';
    } catch (e) {}
  }
  var dest = correosDeDueno_(duenoId);
  var link = APP_.URL + (contactId ? '#c-' + contactId : '');
  var tags = b.tags || (b.contact && b.contact.tags) || [];
  if (typeof tags === 'string') tags = tags.split(',');
  tags = tags.map(function (t) { return String(t).trim().toLowerCase(); });
  if (tipo !== 'cotizacion') {                                        // la cotización la mandó el dueño: no es señal del paciente
    guardarEvento_({ id: Utilities.getUuid().slice(0, 8), c: contactId, n: nombre, t: String(b.phone || (b.contact && b.contact.phone) || ''),
                     tipo: tipo, at: Date.now(), dn: ((leerCache_() || {}).usuarios || {})[duenoId] || '', es: tags.indexOf('es') >= 0, cond: condicionDe_(tags) });
  }
  var n = push_(dest, a[0] + ' · ' + nombre, a[1], link, tipo + '-' + contactId);
  return { ok: true, enviados: n, a: dest.length };
}

/* Dueño de GHL → correos de la app. Sin dueño (o dueño que no está en la app) → a los admins. */
function correosDeDueno_(duenoId) {
  var raw = leerCache_() || {}, nom = norm((raw.usuarios || {})[duenoId] || '');
  var out = [];
  if (nom) Object.keys(EQUIPO_).forEach(function (c) {
    if (EQUIPO_[c].ghl.some(function (g) { return nom.indexOf(g) === 0; })) out.push(c);
  });
  if (!out.length) Object.keys(EQUIPO_).forEach(function (c) { if (EQUIPO_[c].admin) out.push(c); });
  return out;
}

/* ------------------------------------------------------------------ avisos programados (cada 10 min, después de refrescar) */
function avisosProgramados() {
  var raw;
  try { raw = refrescar(); } catch (e) { raw = leerCache_(); if (!raw) throw e; }
  raw.registro = leerRegistro_();
  raw.ahora = Date.now();
  var R = armarRadar(raw), props = PropertiesService.getScriptProperties();
  var av = {}; try { av = JSON.parse(props.getProperty('AVISADOS') || '{}'); } catch (e) { av = {}; }
  var limite = Date.now() - 20 * 86400000;
  Object.keys(av).forEach(function (k) { if (av[k] < limite) delete av[k]; });
  var hora = Number(Utilities.formatDate(new Date(), 'America/Bogota', 'H'));
  var horario = hora >= 7 && hora < 21;                                // de noche no suena nada
  var reglas = { riesgo: ['En riesgo', 'Pasaron 24 h y nadie lo ha llamado'], entrada: ['Sin primer mensaje', 'Lleva más de 2 horas sin que le escribamos'] };
  R.secciones.forEach(function (s) {
    if (!reglas[s.clave] || !horario) return;
    s.items.forEach(function (x) {
      var k = s.clave + ':' + x.opp;
      if (av[k]) return;
      av[k] = Date.now();
      push_(correosDeNombre_(x.dueno), reglas[s.clave][0] + ' · ' + x.nombre, reglas[s.clave][1], APP_.URL + '#c-' + x.contactId, k);
    });
  });
  // resumen de las 8 am
  var hoy = Utilities.formatDate(new Date(), 'America/Bogota', 'yyyy-MM-dd');
  if (hora === 8 && props.getProperty('RESUMEN') !== hoy) {
    props.setProperty('RESUMEN', hoy);
    var tareasHoy = {};                                                 // 14-oct: también las tareas de Pendientes que vencen hoy o están atrasadas
    try {
      if (props.getProperty(PEND_.PROP)) tareasLeer_().forEach(function (t) {
        if (t.estado === 'hecho' || !t.vence || t.vence > hoy) return;
        (t.para === 'Todos' ? equipoNombres_() : [t.para]).forEach(function (n) { tareasHoy[n] = (tareasHoy[n] || 0) + 1; });
      });
    } catch (e) { console.warn('Resumen sin tareas: ' + e.message); }
    Object.keys(EQUIPO_).forEach(function (c) {
      var m = EQUIPO_[c], cuenta = function (s) { return s.items.filter(function (x) { return m.ghl.some(function (g) { return norm(x.dueno).indexOf(g) === 0; }); }).length; };
      var l = R.secciones.map(function (s) { return [s.titulo, cuenta(s)]; }).filter(function (z) { return z[1]; });
      if (tareasHoy[m.nombre]) l.push(['Tareas para hoy', tareasHoy[m.nombre]]);
      push_([c], 'Buenos días, ' + m.nombre, l.length ? l.map(function (z) { return z[1] + ' ' + z[0].toLowerCase(); }).join(' · ') : 'Hoy no tienes pendientes. A vender.', APP_.URL, 'resumen');
    });
  }
  props.setProperty('AVISADOS', JSON.stringify(av));
}

function correosDeNombre_(nombreGhl) {
  var n = norm(nombreGhl), out = [];
  Object.keys(EQUIPO_).forEach(function (c) { if (EQUIPO_[c].ghl.some(function (g) { return n.indexOf(g) === 0; })) out.push(c); });
  if (!out.length) Object.keys(EQUIPO_).forEach(function (c) { if (EQUIPO_[c].admin) out.push(c); });
  return out;
}

/* ------------------------------------------------------------------ instalación (correr una vez desde el editor) */
function instalarApp() {
  var p = PropertiesService.getScriptProperties();
  if (!p.getProperty('HOOK_CLAVE')) p.setProperty('HOOK_CLAVE', Utilities.getUuid().replace(/-/g, ''));
  ScriptApp.getProjectTriggers().forEach(function (t) {
    var f = t.getHandlerFunction();
    if (f === 'refrescar' || f === 'avisosProgramados') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('avisosProgramados').timeBased().everyMinutes(10).create();   // refresca GHL y avisa
  hoja_();
  // la primera pasada solo marca como avisado lo que ya estaba (para no sonar 30 veces de golpe)
  var raw = refrescar(); raw.registro = leerRegistro_(); raw.ahora = Date.now();
  var R = armarRadar(raw), av = {};
  R.secciones.forEach(function (s) { s.items.forEach(function (x) { av[s.clave + ':' + x.opp] = Date.now(); }); });
  p.setProperty('AVISADOS', JSON.stringify(av));
  Logger.log('OK · usuarios GHL: ' + Object.keys(raw.usuarios).map(function (k) { return raw.usuarios[k]; }).join(', '));
  Logger.log('Clave de webhooks lista (no se muestra). ' + R.secciones.map(function (s) { return s.titulo + ': ' + s.items.length; }).join(' · '));
}
